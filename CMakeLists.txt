cmake_minimum_required(VERSION 3.22)
project(OrbitLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ORBITLAB_ENABLE_CUDA "Build CUDA backends" OFF)
option(ORBITLAB_USE_SYSTEM_GTEST "Use installed Google Test" OFF)
include(CTest)
add_library(orbitlab_core INTERFACE)
target_include_directories(orbitlab_core INTERFACE include)
if(ORBITLAB_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(orbitlab_gpu cuda/gpu_simulator.cu)
  target_link_libraries(orbitlab_gpu PUBLIC orbitlab_core CUDA::cudart)
  set_target_properties(orbitlab_gpu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
add_executable(orbitlab src/main.cpp)
target_link_libraries(orbitlab PRIVATE orbitlab_core)
if(ORBITLAB_ENABLE_CUDA)
  target_compile_definitions(orbitlab PRIVATE ORBITLAB_HAS_CUDA=1)
  target_link_libraries(orbitlab PRIVATE orbitlab_gpu)
endif()
if(BUILD_TESTING)
  if(ORBITLAB_USE_SYSTEM_GTEST)
    find_package(GTest REQUIRED)
  else()
    include(FetchContent)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.zip DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(googletest)
  endif()
  add_executable(orbitlab_tests tests/state_test.cpp tests/physics_test.cpp tests/integrator_test.cpp tests/io_test.cpp)
  target_link_libraries(orbitlab_tests PRIVATE orbitlab_core GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(orbitlab_tests)
  if(ORBITLAB_ENABLE_CUDA)
    add_executable(orbitlab_cuda_tests tests/cuda_force_test.cpp tests/cuda_integrator_test.cpp tests/cuda_tiling_test.cpp)
    target_include_directories(orbitlab_cuda_tests PRIVATE cuda)
    target_link_libraries(orbitlab_cuda_tests PRIVATE orbitlab_gpu GTest::gtest_main)
    gtest_discover_tests(orbitlab_cuda_tests)
  endif()
endif()
